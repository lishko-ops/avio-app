const TIME_SLOTS = ['9:00 AM', '10:30 AM', '12:00 PM', '1:30 PM', '3:00 PM', '4:30 PM', '6:00 PM', '7:30 PM'];

function nextDays(n) {
  const out = [];
  const base = new Date();
  for (let i = 0; i < n; i++) {
    const d = new Date(base);
    d.setDate(base.getDate() + i);
    out.push(d);
  }
  return out;
}

function money(n) {
  return n.toLocaleString('en-US', { style: 'currency', currency: 'USD' });
}

                setBalance((b) => b + 100);

            </button>

                key={t}

              >
                {t}
              </button>

          </div>

            <Sparkles size={14} /> AI suggest my plan

          <p style={{ fontSize: 11, color: THEME.muted, marginTop: 18, marginBottom: 8 }}>ALLOCATE BY CATEGORY</p>

                <button onClick={() => setAllocations((a) => ({ ...a, [cat]: Math.max(0, a[cat] - 25) }))} style={{ width: 22, height: 22, borderRadius: '50%', border: `1px solid ${THEME.paperLine}`, display: 'flex', alignItems: 'center', justifyContent: 'center' }}>

                </button>

                  <Plus size={12} />
                </button>
              </div>
            </div>
          ))}

            <span style={{ fontFamily: 'JetBrains Mono, monospace', fontSize: 13, fontWeight: 600, color: THEME.textLight }}>{money(Math.max(0, monthlyBudget - spent))}</span>

        </div>

        <div style={{ height: '100%', overflowY: 'auto', background: THEME.paper, padding: `calc(env(safe-area-inset-top) + 18px) 18px calc(env(safe-area-inset-bottom) + 90px)` }}>

              <div key={b.id} style={{ background: '#fff', border: `1px solid ${THEME.paperLine}`, borderRadius: 14, padding: 14 }}>
                <div className="flex items-center justify-between">
                  <p style={{ fontFamily: 'Space Grotesk, sans-serif', fontWeight: 600, fontSize: 14, color: THEME.textLight }}>{b.name}</p>

            <button key={t.id} onClick={() => { setTab(t.id); setActiveCat(null); }} className="flex flex-col items-center gap-1">

                const active = i === selDate;

                  <button key={i} onClick={() => setSelDate(i)} style={{ minWidth: 44, padding: '8px 0', borderRadius: 10, background: active ? THEME.coral : THEME.paper, color: active ? '#fff' : THEME.textLight, textAlign: 'center', flexShrink: 0 }}>
                    <div style={{ fontSize: 10 }}>{d.toLocaleDateString('en-US', { weekday: 'short' })}</div>
                    <div style={{ fontSize: 14, fontWeight: 600 }}>{d.getDate()}</div>
                  </button>

              })}

            <p style={{ fontSize: 11, color: THEME.muted, marginTop: 16, marginBottom: 8, letterSpacing: 0.5 }}>CHOOSE A TIME</p>

                const active = i === selTime;
                return (
                  <button key={t} onClick={() => setSelTime(i)} style={{ padding: '8px 0', borderRadius: 10, fontSize: 12, border: `1px solid ${active ? THEME.coral : THEME.paperLine}`, background: active ? THEME.coral : '#fff', color: active ? '#fff' : THEME.textLight }}>
                    {t}
                  </button>
                );
              })}
            </div>

            <div style={{ marginTop: 18, borderTop: `1px solid ${THEME.paperLine}`, paddingTop: 12 }}>
              <div className="flex items-center justify-between">
                <span style={{ fontSize: 12, color: THEME.muted }}>Service price</span>
                <span style={{ fontSize: 12, color: THEME.textLight }}>{money(bookingTarget.price)}</span>
              </div>
              <div className="flex items-center justify-between mt-1">
                <span style={{ fontSize: 12, color: THEME.muted }}>Tax (8%)</span>
                <span style={{ fontSize: 12, color: THEME.textLight }}>{money(Math.round(bookingTarget.price * 0.08 * 100) / 100)}</span>
              </div>
              <div className="flex items-center justify-between mt-2" style={{ paddingTop: 8, borderTop: `1px solid ${THEME.paperLine}` }}>
                <span style={{ fontSize: 13, fontWeight: 600, color: THEME.textLight }}>Total</span>
                <span style={{ fontFamily: 'JetBrains Mono, monospace', fontSize: 15, fontWeight: 600, color: THEME.textLight }}>{money(Math.round(bookingTarget.price * 1.08 * 100) / 100)}</span>
              </div>
            </div>

            <div className="flex gap-3 mt-4">
              <button onClick={() => setBookingTarget(null)} style={{ flex: 1, padding: '11px 0', borderRadius: 24, border: `1px solid ${THEME.paperLine}`, color: THEME.textLight, fontSize: 13, fontWeight: 500 }}>Cancel</button>
              <button onClick={confirmBooking} disabled={selTime == null} style={{ flex: 2, padding: '11px 0', borderRadius: 24, background: selTime == null ? THEME.paperLine : THEME.coral, color: selTime == null ? THEME.muted : '#fff', fontSize: 13, fontWeight: 600 }}>
                Confirm and pay
              </button>
            </div>
          </div>
        </div>
      )}

      {/* toast */}
      {toast && (
        <div style={{ position: 'absolute', top: 'calc(env(safe-area-inset-top) + 54px)', left: 12, right: 12, background: THEME.textLight, color: '#fff', fontSize: 12, padding: '10px 14px', borderRadius: 12, zIndex: 20, boxShadow: '0 8px 20px rgba(0,0,0,0.25)' }}>
          {toast}
        </div>
      )}
    </div>
  );
}
